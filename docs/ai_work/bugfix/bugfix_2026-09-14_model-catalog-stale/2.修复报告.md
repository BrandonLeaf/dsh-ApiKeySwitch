+  function effectiveRoute(profile, providers, fallback) {
+    const provider = profile.provider || fallback.provider || DEFAULT_PROVIDER
+    const entry = providers.find((p) => p.id === provider)
+    const ids = entry ? entry.models.map((m) => m.id) : []
+    const stored = profile.model || ''
+    let model = stored
+    let stale = false
+    if (stored !== '' && ids.length > 0 && ids.indexOf(stored) === -1) { stale = true; model = '' }

+    }
+    return { provider, model, stale }
+  }

## 编译影响

1. 本项目无构建步骤（无 prepare 脚本，`build-test.no-build` 约束），源码即运行产物；`lib/` 与 `plugin/` 均通过 `node --check` 语法校验（`plugin/*.js` 为函数体，校验时包一层 `(function(){ ... })`）。
2. 部署形态需要把 `lib/` 与 `package.json` 复制到 `~/.dsh/profiles/web/node_modules/apikeyswitch/` 并重启 `dsh web` 才会加载新的 Host 半区；Client 半区在页面刷新时按 `apikeyswitch/client.js` 重新加载。
3. 依赖与接口边界：无新增 npm 依赖；新增使用的宿主服务为 `llm`、`settings`、`agentDefaultModel`（全部为 web 组合已挂载服务），缺失时按 `undefined` 降级（供应商回落 `deepseek-official`、模型回落空值，不抛错）。
