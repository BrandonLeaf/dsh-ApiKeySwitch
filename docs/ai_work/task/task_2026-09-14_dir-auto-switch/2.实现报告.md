+    try {
+      const picker = ctx.get('directoryPicker')
+      const capability = picker && typeof picker.capability === 'function' ? picker.capability() : null
+      if (capability && typeof capability.kind === 'string' && capability.kind !== '') return capability.kind

+  async function pickDirectory(res) {
+    const kind = pickerKind()
+    if (kind !== 'native') return { ok: false, error: '当前部署未提供原生目录选择器（directoryPicker capability=' + kind + '），请手动填写绝对路径。' }
+    const capability = ctx.get('directoryPicker').capability()
+    const controller = new AbortController()
+    const onClose = () => { controller.abort() }
+    try { if (res && typeof res.on === 'function') res.on('close', onClose) } catch (e) { /* 无响应对象时由选择框自身生命周期结束 */ }

## 已知边界

1. 目录自动切换受运行会话守卫约束：有**其他**会话正在执行任务时会被拒绝（`guard:true`），界面提示「已阻止目录自动切换」，下次激活重试；需要立即切换时先停止运行中的会话。
2. 目录绑定优先于会话记忆：同一会话内手动切到其他项目后，再次激活该会话仍会按目录绑定切回。
3. 目录匹配依赖宿主 Canonic 化能力：`workspaceRegistry.resolveByPath()` 在目录当前不存在时抛错，此时按字符串比较；若用户配置了指向不存在目录的绑定，该绑定不会命中任何会话。
4. 系统目录选择框在 dsh 宿主机屏幕弹出：dsh 绑定非 loopback、经 SSH 启动或无显示会话时宿主挂载 browse 后端，插件隐藏「选择」按钮；此路径不做浏览器内逐级浏览界面，仍可手动填写绝对路径
5. 自动化测试未点击真实系统对话框（会弹出 macOS 模态框），采用 fetch 拦截验证按钮到 RPC 到输入框的完整链路；对话框本身复用应用「添加工作区」的同一 capability，未做二次实现
6. 动态插件形态的 Client 半区未在 DSH 0.1.5-alpha.2 实例中实测（该实例未挂载动态插件运行器入口），仅完成语法校验与与部署版一致的逻辑同步。
